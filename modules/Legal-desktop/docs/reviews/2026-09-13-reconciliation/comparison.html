+        hashlib.sha256,
+    ).hexdigest()
+    return hmac.compare_digest(supplied, expected)

+function selectorBlock(source, selector) {

+  for (let index = openAt; index < source.length; index += 1) {
+    if (source[index] === "{") depth += 1;
+    if (source[index] === "}") depth -= 1;
+    if (depth === 0) return source.slice(openAt + 1, index);
+  }
